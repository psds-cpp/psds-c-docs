    \begin{myitemize}

        \item opaque pointer — сокрытие представления;

        \item handle — число вместо указателя;

        \item init/deinit — жизненный цикл объекта;

        \item options struct — конфигурация как данные;

        \item возврат ошибок — коды и out-параметры;

        \item function table — поведение как данные;

        \item tagged union — «вариант» на C.

    \end{myitemize}

\begin{frame}

    \frametitle{Из чего складывается проектирование на C}

    \begin{myitemize}

        \item В C нет классов, наследования и исключений —
            но крупные проекты на нём устроены «по-инженерному».

        \item Проектирование на C собирают из простых механизмов:
            \texttt{struct}, указатели, указатели на функции, \texttt{enum}.

        \item Идиома — устойчивое сочетание этих механизмов,
            которое повторяется из проекта в проект.

        \item Программа лекции: семь идиом, в порядке от простых к сложным.

    \end{myitemize}

\end{frame}

\begin{frame}[fragile]

    \frametitle{Opaque pointer: сокрытие реализации}

    \begin{columns}[T]

        \begin{column}{0.55\textwidth}

            \begin{myitemize}

                \item Потребитель видит только указатель \texttt{sec\_t*}
                    и объявления функций.

                \item Определение \texttt{struct} — только в файле .c
                    (реализации).

                \item Заголовок: перекрёстное объявление
                    + объявления API.

                \item Представление можно менять без пересборки клиентов.

            \end{myitemize}

        \end{column}

        \begin{column}{0.4\textwidth}

            \begin{myinplacelisting}[minted language=c]
struct secret;
typedef struct secret sec_t;
            \end{myinplacelisting}

        \end{column}

    \end{columns}

\end{frame}

\begin{frame}[fragile]

    \frametitle{Handle: число вместо указателя}

    \begin{columns}[T]

        \begin{column}{0.55\textwidth}

            \begin{myitemize}
            \setlength{\itemsep}{0.25ex}

                \item Handle — это индекс объекта в таблице реализации:
                    потребитель получает число, а не указатель.

                \item Развитие opaque pointer: представление по-прежнему
                    скрыто, но вместо указателя — обычное целое.

                \item Плюсы: легко передавать, хранить и сравнивать;
                    «висящих» указателей на стороне клиента не бывает.

                \item Реализация: пул объектов + поиск свободного места
                    + невалидное значение.

            \end{myitemize}

        \end{column}

        \begin{column}{0.4\textwidth}

            \begin{myinplacelisting}[minted language=c]
typedef uint32_t handle_t;
#define BAD ((handle_t)-1)
            \end{myinplacelisting}

        \end{column}

    \end{columns}

\end{frame}

\begin{frame}[fragile]

    \frametitle{Init/deinit: управление временем жизни}

    \begin{columns}[T]

        \begin{column}{0.5\textwidth}

            \begin{myitemize}
            \setlength{\itemsep}{1ex}

                \item Пара функций: \texttt{init} создаёт и
                    инициализирует объект, \texttt{deinit} освобождает
                    ресурсы.

                \item \texttt{init} может провалиться — вернёт
                    \texttt{NULL}; результат \texttt{init}
                    всегда проверяем.

                \item \texttt{deinit} освобождает всё, что занял
                    \texttt{init}: память, файлы, соединения.

                \item Аналог конструктора/деструктора в C: RAII нет —
                    дисциплина ручная.

            \end{myitemize}

        \end{column}

        \begin{column}{0.46\textwidth}

            \begin{myinplacelisting}[minted language=c]
struct cfg *cfg_init(int cap);
void cfg_deinit(struct cfg *c);
            \end{myinplacelisting}

        \end{column}

    \end{columns}

\end{frame}

\begin{frame}[fragile]

    \frametitle{Options struct: конфигурация как данные}

    \begin{columns}[T]

        \begin{column}{0.55\textwidth}

            \begin{myitemize}
            \setlength{\itemsep}{0.25ex}

                \item Много параметров у функции (особенно опциональных)
                    — выносим их в структуру конфигурации.

                \item 0 / «ноль» — типичное значение по умолчанию:
                    «не задано».

                \item Расширение API — новое поле в структуре;
                    старые вызовы не ломаются.

                \item Конфигуцию удобно передавать «пакетом»,
                    копировать, хранить.

                \item Продолжение предыдущей темы:
                    \texttt{log\_init(const log\_opts\_t *opts)}
                    вместо \texttt{log\_init(int level, int timeout, …)}.

            \end{myitemize}

        \end{column}

        \begin{column}{0.4\textwidth}

            \begin{myinplacelisting}[minted language=c]
typedef struct {
    int level;
    int timeout;
} log_opts_t;
            \end{myinplacelisting}

        \end{column}

    \end{columns}

\end{frame}

\begin{frame}[fragile]

    \frametitle{Возврат ошибок: подходы}

    \begin{columns}[T]

        \begin{column}{0.55\textwidth}

            \begin{myitemize}
            \setlength{\itemsep}{0.25ex}

                \item Исключений в C нет: ошибка возвращается явно.

                \item Код возврата: 0 = OK, > 0 — код ошибки;
                    явно и самоочевидно.

                \item errno-стиль: вернуть \texttt{-1}/\texttt{NULL},
                    код — в глобальной переменной \texttt{errno}
                    (POSIX); слабость: состояние общее для потока.

                \item Статус + out-параметр: результат — отдельный
                    канал, а не «ошибка».

                \item Выбор: собственный API — явные коды;
                    POSIX-совместимость — \texttt{errno}.

            \end{myitemize}

        \end{column}

        \begin{column}{0.4\textwidth}

            \begin{myinplacelisting}[minted language=c]
typedef enum {
    ERR_OK = 0,
    ERR_EMPTY,
} err_t;
            \end{myinplacelisting}

        \end{column}

    \end{columns}

\end{frame}

\begin{frame}[fragile]

    \frametitle{Function table: поведение как данные}

    \begin{columns}[T]

        \begin{column}{0.52\textwidth}

            \begin{myitemize}

                \item Struct из указателей на функции — таблица операций
                    («vtable» на C).

                \item Объект = данные + указатель на свою
                    таблицу операций.

                \item Разные «классы» — разные таблицы
                    с общими подписями.

                \item Вызовы идут единообразно через таблицу:
                    \texttt{d->ops->open(d)} — без if/switch по типу.

            \end{myitemize}

        \end{column}

        \begin{column}{0.46\textwidth}

            \begin{myinplacelisting}[minted language=c]
struct dev;
struct ops {
    void (*open)(struct dev *d);
};
            \end{myinplacelisting}

        \end{column}

    \end{columns}

\end{frame}

\begin{frame}[fragile]

    \frametitle{Tagged union: «вариант» на C}

    \begin{columns}[T]

        \begin{column}{0.5\textwidth}

            \begin{myitemize}
            \setlength{\itemsep}{1ex}

                \item «Вариант» на C: \texttt{enum} (тег) + \texttt{union}
                    (данные разных типов).

                \item В любой момент действителен ровно один член
                    union — тег показывает, какой.

                \item Инвариант: тег и поле выставляются вместе.

                \item Геттер без проверки тега — undefined behavior;
                    поэтому геттер возвращает ошибку.

                \item Где применяется: значения настроек, узлы AST,
                    результат нескольких типов.

            \end{myitemize}

        \end{column}

        \begin{column}{0.44\textwidth}

            \begin{myinplacelisting}[minted language=c,
                minted options={fontsize=\scriptsize, breaklines,
                    breakanywhere, linenos, escapeinside=||,
                    numbersep=2pt}]
enum kind { K_INT, K_STR };
struct val {
    enum kind k;
    union { int i; const char *s; } u;
};
            \end{myinplacelisting}

        \end{column}

    \end{columns}

\end{frame}

\begin{frame}

    \frametitle{Что получили}

    \begin{myitemize}

        \item Инкапсуляция: opaque pointer → handle.

        \item Жизненный цикл: init/deinit; конфигурация:
            options struct.

        \item Ошибки: коды возврата, out-параметры, errno.

        \item Поведение: function table («vtable» на C).

        \item Данные: tagged union («вариант» на C).

        \item Идиомы комбинируются: API библиотеки на C можно
            устроить «как класс».

    \end{myitemize}

\end{frame}
